Here's a simple C++ implementation of a single-layer feedforward neural network (perceptron) with training using the perceptron learning rule. This example learns the logical AND function.

#include <iostream>
#include <vector>
#include <random>

class Perceptron {
private:
    std::vector<double> weights;
    double bias;
    double learningRate;

    int activation(double x) {
        return (x >= 0.0) ? 1 : 0;
    }

public:
    Perceptron(int numInputs, double lr = 0.1)
        : weights(numInputs), bias(0.0), learningRate(lr)
    {
        std::random_device rd;
        std::mt19937 gen(rd());
        std::uniform_real_distribution<> dist(-1.0, 1.0);

        for (auto& w : weights)
            w = dist(gen);

        bias = dist(gen);
    }

    int predict(const std::vector<double>& inputs) {
        double sum = bias;

        for (size_t i = 0; i < inputs.size(); i++) {
            sum += inputs[i] * weights[i];
        }

        return activation(sum);
    }

    void train(const std::vector<std::vector<double>>& trainingInputs,
               const std::vector<int>& labels,
               int epochs)
    {
        for (int epoch = 0; epoch < epochs; epoch++) {
            for (size_t i = 0; i < trainingInputs.size(); i++) {
                int prediction = predict(trainingInputs[i]);
                int error = labels[i] - prediction;

                // Update weights
                for (size_t j = 0; j < weights.size(); j++) {
                    weights[j] += learningRate * error * trainingInputs[i][j];
                }

                // Update bias
                bias += learningRate * error;
            }
        }
    }

    void printParameters() const {
        std::cout << "Weights: ";
        for (double w : weights)
            std::cout << w << " ";

        std::cout << "\nBias: " << bias << "\n";
    }
};

int main() {
    // AND gate dataset
    std::vector<std::vector<double>> inputs = {
        {0, 0},
        {0, 1},
        {1, 0},
        {1, 1}
    };

    std::vector<int> outputs = {
        0,
        0,
        0,
        1
    };

    Perceptron p(2, 0.1);

    p.train(inputs, outputs, 20);

    p.printParameters();

    std::cout << "\nPredictions:\n";

    for (size_t i = 0; i < inputs.size(); i++) {
        std::cout << inputs[i][0] << " AND "
                  << inputs[i][1] << " = "
                  << p.predict(inputs[i]) << "\n";
    }

    return 0;
}

Below is a minimal but complete multilayer feedforward neural network (MLP) in modern C++ using:

Sigmoid activation
Backpropagation
Gradient descent
One hidden layer
Training on XOR (which a single perceptron cannot learn)
#include <iostream>
#include <vector>
#include <cmath>
#include <random>

class NeuralNetwork {
private:
    int inputSize;
    int hiddenSize;
    int outputSize;
    double learningRate;

    std::vector<std::vector<double>> W1;
    std::vector<std::vector<double>> W2;
    std::vector<double> B1;
    std::vector<double> B2;

    static double sigmoid(double x) {
        return 1.0 / (1.0 + std::exp(-x));
    }

    static double sigmoidDerivative(double y) {
        return y * (1.0 - y);
    }

    double randomWeight() {
        static std::random_device rd;
        static std::mt19937 gen(rd());
        static std::uniform_real_distribution<> dist(-1.0, 1.0);
        return dist(gen);
    }

public:
    NeuralNetwork(int in, int hidden, int out, double lr = 0.1)
        : inputSize(in),
          hiddenSize(hidden),
          outputSize(out),
          learningRate(lr)
    {
        W1.resize(hiddenSize, std::vector<double>(inputSize));
        W2.resize(outputSize, std::vector<double>(hiddenSize));

        B1.resize(hiddenSize);
        B2.resize(outputSize);

        for (auto& row : W1)
            for (double& w : row)
                w = randomWeight();

        for (auto& row : W2)
            for (double& w : row)
                w = randomWeight();

        for (double& b : B1)
            b = randomWeight();

        for (double& b : B2)
            b = randomWeight();
    }

    std::vector<double> predict(const std::vector<double>& input) {
        std::vector<double> hidden(hiddenSize);
        std::vector<double> output(outputSize);

        // Hidden layer
        for (int h = 0; h < hiddenSize; h++) {
            double sum = B1[h];

            for (int i = 0; i < inputSize; i++)
                sum += input[i] * W1[h][i];

            hidden[h] = sigmoid(sum);
        }

        // Output layer
        for (int o = 0; o < outputSize; o++) {
            double sum = B2[o];

            for (int h = 0; h < hiddenSize; h++)
                sum += hidden[h] * W2[o][h];

            output[o] = sigmoid(sum);
        }

        return output;
    }

    void train(const std::vector<double>& input,
               const std::vector<double>& target)
    {
        // ===== Forward Pass =====

        std::vector<double> hidden(hiddenSize);
        std::vector<double> output(outputSize);

        for (int h = 0; h < hiddenSize; h++) {
            double sum = B1[h];

            for (int i = 0; i < inputSize; i++)
                sum += input[i] * W1[h][i];

            hidden[h] = sigmoid(sum);
        }

        for (int o = 0; o < outputSize; o++) {
            double sum = B2[o];

            for (int h = 0; h < hiddenSize; h++)
                sum += hidden[h] * W2[o][h];

            output[o] = sigmoid(sum);
        }

        // ===== Backpropagation =====

        std::vector<double> outputDelta(outputSize);

        for (int o = 0; o < outputSize; o++) {
            double error = target[o] - output[o];
            outputDelta[o] = error * sigmoidDerivative(output[o]);
        }

        std::vector<double> hiddenDelta(hiddenSize);

        for (int h = 0; h < hiddenSize; h++) {
            double error = 0.0;

            for (int o = 0; o < outputSize; o++)
                error += outputDelta[o] * W2[o][h];

            hiddenDelta[h] = error * sigmoidDerivative(hidden[h]);
        }

        // ===== Gradient Descent Update =====

        for (int o = 0; o < outputSize; o++) {
            for (int h = 0; h < hiddenSize; h++) {
                W2[o][h] += learningRate *
                            outputDelta[o] *
                            hidden[h];
            }

            B2[o] += learningRate * outputDelta[o];
        }

        for (int h = 0; h < hiddenSize; h++) {
            for (int i = 0; i < inputSize; i++) {
                W1[h][i] += learningRate *
                            hiddenDelta[h] *
                            input[i];
            }

            B1[h] += learningRate * hiddenDelta[h];
        }
    }
};

int main() {
    NeuralNetwork nn(2, 4, 1, 0.5);

    std::vector<std::vector<double>> X = {
        {0,0},
        {0,1},
        {1,0},
        {1,1}
    };

    std::vector<std::vector<double>> Y = {
        {0},
        {1},
        {1},
        {0}
    };

    // Train
    for (int epoch = 0; epoch < 10000; epoch++) {
        for (size_t i = 0; i < X.size(); i++) {
            nn.train(X[i], Y[i]);
        }
    }

    // Test
    std::cout << "XOR Results\n";

    for (size_t i = 0; i < X.size(); i++) {
        auto out = nn.predict(X[i]);

        std::cout
            << X[i][0] << " XOR "
            << X[i][1] << " = "
            << out[0]
            << std::endl;
    }

    return 0;
}
